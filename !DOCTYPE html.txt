            document.querySelectorAll('.layer-btn').forEach(b => b.classList.remove('active'));
            if(btnElement) btnElement.classList.add('active');

            // Find Data
            const layer = architectureLayers.find(l => l.id === id);
            if (!layer) return;

            // Update Panel
            document.getElementById('detail-title').textContent = layer.name;
            const packageBadge = document.getElementById('detail-package');
            packageBadge.textContent = layer.tag;
            packageBadge.classList.remove('hidden');

            let contentHTML = `
                <div class="w-full fade-in h-full flex flex-col">
                    <p class="text-slate-700 mb-6 text-lg">${layer.desc}</p>
                    
                    <div class="grid grid-cols-1 md:grid-cols-2 gap-6 flex-grow">
                        <div>
                            <h4 class="text-xs uppercase tracking-wider font-bold text-slate-500 mb-3">Key Modules</h4>
                            <ul class="space-y-2">
                                ${layer.modules.map(m => `
                                    <li class="flex items-center text-sm font-mono text-slate-700 bg-slate-50 px-3 py-2 rounded border border-slate-100">
                                        <span class="text-brand-500 mr-2 text-xs">📄</span> ${m}
                                    </li>
                                `).join('')}
                            </ul>
                            ${layer.visual || ''}
                        </div>
            `;

            if (layer.example) {
                let codeClass = layer.example.type === 'json' ? 'text-blue-400' : 'text-emerald-400';
                contentHTML += `
                        <div class="flex flex-col h-full">
                            <h4 class="text-xs uppercase tracking-wider font-bold text-slate-500 mb-3">${layer.example.title}</h4>
                            <div class="bg-slate-900 rounded-lg p-4 flex-grow overflow-auto shadow-inner border border-slate-800">
                                <pre class="font-mono text-sm leading-relaxed ${codeClass}"><code>${layer.example.code}</code></pre>
                            </div>
                        </div>
                `;
            } else {
                 contentHTML += `
                        <div class="flex items-center justify-center bg-slate-50 rounded-lg border border-dashed border-slate-300 p-6">
                            <span class="text-slate-400 text-sm italic">Structural Core Layer</span>
                        </div>
                `;
            }

            contentHTML += `</div></div>`;
            document.getElementById('detail-content').innerHTML = contentHTML;
        }

        function initAgentsList() {
            const list = document.getElementById('agent-list');
            agentData.forEach(agent => {
                const div = document.createElement('div');
                div.className = `flex justify-between items-center p-3 rounded-lg border ${agent.color}`;
                div.innerHTML = `
                    <div>
                        <span class="font-bold text-sm block">${agent.name}</span>
                        <span class="text-xs opacity-80">${agent.role}</span>
                    </div>
                    <div class="text-right">
                        <span class="font-bold text-lg leading-none">${agent.sections}</span>
                        <span class="text-[10px] uppercase block opacity-80">Sections</span>
                    </div>
                `;
                list.appendChild(div);
            });
        }

        function initAgentChart() {
            const ctx = document.getElementById('agentChart').getContext('2d');
            
            // Extract data for chart
            const labels = agentData.map(a => a.name);
            const data = agentData.map(a => a.sections);
            const backgroundColors = [
                'rgba(59, 130, 246, 0.7)',  // Blue
                'rgba(16, 185, 129, 0.7)',  // Green
                'rgba(168, 85, 247, 0.7)',  // Purple
                'rgba(249, 115, 22, 0.7)',  // Orange
                'rgba(236, 72, 153, 0.7)',  // Pink
                'rgba(20, 184, 166, 0.7)'   // Brand (Teal)
            ];
            const borderColors = backgroundColors.map(c => c.replace('0.7', '1'));

            new Chart(ctx, {
                type: 'bar',
                data: {
                    labels: labels,
                    datasets: [{
                        label: 'Accessible UI Sections',
                        data: data,
                        backgroundColor: backgroundColors,
                        borderColor: borderColors,
                        borderWidth: 1,
                        borderRadius: 4
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false, // CRITICAL: Allows CSS to control height
                    plugins: {
                        legend: { display: false },
                        tooltip: {
                            callbacks: {
                                label: function(context) {
                                    return `Controls ${context.raw} Section${context.raw > 1 ? 's' : ''}`;
                                }
                            }
                        }
                    },
                    scales: {
                        y: {
                            beginAtZero: true,
                            ticks: { stepSize: 1 },
                            title: { display: true, text: 'Number of DOM Sections', color: '#64748b' }
                        },
                        x: {
                            ticks: { 
                                maxRotation: 45, 
                                minRotation: 45,
                                color: '#475569',
                                // Label wrapping logic (16-char) implemented by truncating/formatting
                                callback: function(value, index, values) {
                                    let label = labels[index];
                                    if(label.length > 16) return label.substr(0,14) + '..';
                                    return label;
                                }
                            }
                        }
                    }
                }
            });
        }

        function switchWorkflow(type) {
            const devFlow = document.getElementById('flow-dev');
            const runtimeFlow = document.getElementById('flow-runtime');
            const btnDev = document.getElementById('btn-flow-dev');
            const btnRuntime = document.getElementById('btn-flow-runtime');

            if (type === 'dev') {
                devFlow.classList.remove('hidden');
                runtimeFlow.classList.add('hidden');
                
                btnDev.classList.add('bg-white', 'shadow-sm', 'text-brand-600');
                btnDev.classList.remove('text-slate-600');
                
                btnRuntime.classList.remove('bg-white', 'shadow-sm', 'text-brand-600');
                btnRuntime.classList.add('text-slate-600');
            } else {
                devFlow.classList.add('hidden');
                runtimeFlow.classList.remove('hidden');
                
                btnRuntime.classList.add('bg-white', 'shadow-sm', 'text-brand-600');
                btnRuntime.classList.remove('text-slate-600');
                
                btnDev.classList.remove('bg-white', 'shadow-sm', 'text-brand-600');
                btnDev.classList.add('text-slate-600');
            }
        }

            initAgentChart();
        });
